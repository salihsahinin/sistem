```sql
CREATE VIEW IF NOT EXISTS v_kosum_tahmin_sapma AS
WITH esli AS (
  SELECT k.kosum_id, k.wf_id, k.aile, k.rol, k.modeller, k.output_tokens, k.arac_cagrisi, k.sure_sn,
         t.tahmin_id, t.rol_kod, t.hedef_model, t.hedef_effort, t.emir_dosya, t.emir_satir,

         COALESCE(t.cikti_token_alt, t.formul_carpani_alt * k.arac_cagrisi) AS cikti_alt,
         COALESCE(t.cikti_token_ust, t.formul_carpani_ust * k.arac_cagrisi) AS cikti_ust
  FROM amb_kopru_kosum_tahmin b
  JOIN amb_olgu_kosum  k ON k.kosum_id  = b.kosum_id
  JOIN amb_olgu_tahmin t ON t.tahmin_id = b.tahmin_id
)
SELECT wf_id, aile, rol_kod, hedef_model, hedef_effort, kosum_id, tahmin_id,
       cikti_alt, cikti_ust, output_tokens AS cikti_gercek,
       CASE WHEN output_tokens > cikti_ust THEN 'AŞTI'
            WHEN output_tokens < cikti_alt THEN 'ALTINDA'
            ELSE 'BANT-İÇİ' END                                            AS cikti_hukmu,
       ROUND(1.0 * output_tokens / NULLIF(cikti_ust, 0), 2)                AS cikti_ust_orani,
       arac_alt, arac_ust, arac_cagrisi AS arac_gercek,
       CASE WHEN arac_cagrisi > arac_ust THEN 'AŞTI'
            WHEN arac_cagrisi < arac_alt THEN 'ALTINDA' ELSE 'BANT-İÇİ' END AS arac_hukmu,
       sure_dk_alt, sure_dk_ust, ROUND(sure_sn / 60.0, 1)                   AS sure_dk_gercek,
       CASE WHEN sure_sn / 60.0 > sure_dk_ust THEN 'AŞTI'
            WHEN sure_sn / 60.0 < sure_dk_alt THEN 'ALTINDA' ELSE 'BANT-İÇİ' END AS sure_hukmu,
       tahminci_model, formul, emir_dosya || ':' || emir_satir              AS tahmin_kaynagi

ORDER BY wf_id DESC, aile, rol_kod;
```

### `v_skill_kosum`

```sql
CREATE VIEW IF NOT EXISTS v_skill_kosum AS
-- KANAL 1: ana-sıra skill çağrıları (hareket; skill adı + model + FAIL var, token/süre YOK — bkz. rapor §C.4)
SELECT 'hareket'                                                            AS kaynak,
       substr(h.hedef, 1, CASE WHEN instr(h.hedef,' ')>0
                               THEN instr(h.hedef,' ')-1 ELSE length(h.hedef) END) AS skill,
       h.model                                                              AS model,
       CASE WHEN h.model LIKE '%fable%' THEN 'ust' ELSE 'alt' END           AS model_katmani,
       COUNT(*)                                                             AS n,
       SUM(CASE WHEN h.sonuc = 'hata' THEN 1 ELSE 0 END)                    AS fail_n,
       NULL AS output_tokens, NULL AS sure_sn_toplam,                       -- [BOŞLUK] ölçüm yok
       MIN(h.tarih_tr) AS ilk, MAX(h.tarih_tr) AS son
FROM hareket h
WHERE h.arac = 'Skill'
GROUP BY skill, h.model
UNION ALL
-- KANAL 2: alt-ajana delege edilmiş skill koşumları (token/süre VAR; skill adı bugün YOK → SH-1 damgası şart)
SELECT 'ajan' AS kaynak, k.skill_ad AS skill, k.modeller AS model,
       CASE WHEN k.modeller LIKE '%fable%' THEN 'ust' ELSE 'alt' END,
       COUNT(*), SUM(CASE WHEN k.sonuc_kod = 'FAIL' THEN 1 ELSE 0 END),
       SUM(k.output_tokens), SUM(k.sure_sn),
       MIN(date(k.valid_time_bas)), MAX(date(k.valid_time_bas))
FROM amb_olgu_kosum k
WHERE k.skill_ad IS NOT NULL          -- ← bugün HİÇBİR satır sağlamıyor (rapor §C.3)
```
